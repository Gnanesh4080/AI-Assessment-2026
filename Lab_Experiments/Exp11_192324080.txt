# Map Coloring using CSP (Backtracking)
# Graph representing neighboring regions
graph = {
    'A': ['B', 'C', 'D'],
    'B': ['A', 'C'],
    'C': ['A', 'B', 'D'],
    'D': ['A', 'C']
}
# Available colors
colors = ['Red', 'Green', 'Blue']

# Dictionary to store assigned colors
color_assignment = {}

# Function to check if color assignment is valid
def is_safe(region, color):
    for neighbor in graph[region]:
        if neighbor in color_assignment and color_assignment[neighbor] == color:
            return False
    return True

# Backtracking function
def map_coloring(regions):
    if len(color_assignment) == len(regions):
        return True

    region = regions[len(color_assignment)]

    for color in colors:
        if is_safe(region, color):
            color_assignment[region] = color

            if map_coloring(regions):
                return True

            del color_assignment[region]

    return False

regions = list(graph.keys())

if map_coloring(regions):
    print("Color Assignment:")
    for region in regions:
        print(region, "->", color_assignment[region])
else:
    print("No solution exists.")
